  <script>
    const FORMSPREE_ENDPOINT = "YOUR_FORMSPREE_ENDPOINT_HERE";

    let selectedDate = "";
    let selectedActivity = "";
    let yesBtnScale = 1;

    const noBtnMessages = [
      "Are you sure?",
      "Better say yes!",
      "Please say yes 🙏",
      "Don't do this!",
      "Heart broken :("
    ];
    let noClickCount = 0;

    // Background floating hearts animation logic
    const heartsContainer = document.getElementById('hearts-container');
    for (let i = 0; i < 15; i++) {
      const heart = document.createElement('div');
      heart.classList.add('heart');
      heart.innerText = '💗';
      heart.style.left = `${Math.random() * 100}%`;
      heart.style.animationDelay = `${Math.random() * 5}s`;
      heart.style.animationDuration = `${4 + Math.random() * 4}s`;
      heartsContainer.appendChild(heart);
    }

    const noBtn = document.getElementById('no-btn');
    const yesBtn = document.getElementById('yes-btn');

    noBtn.addEventListener('click', () => {
      if (noClickCount < noBtnMessages.length) {
        noBtn.innerText = noBtnMessages[noClickCount];
        noClickCount++;
      }
      yesBtnScale += 0.25;
      yesBtn.style.transform = `scale(${yesBtnScale})`;
    });

    yesBtn.addEventListener('click', () => {
      document.getElementById('step-1').classList.add('hidden');
      document.getElementById('step-2').classList.remove('hidden');
    });

    document.getElementById('next-to-step3').addEventListener('click', () => {
      selectedDate = document.getElementById('date-picker').value;
      if (!selectedDate) {
        alert("Please pick a date!");
        return;
      }
      document.getElementById('step-2').classList.add('hidden');
      document.getElementById('step-3').classList.remove('hidden');
    });

    function selectActivity(element, activity) {
      document.querySelectorAll('.option-btn').forEach(btn => btn.classList.remove('selected'));
      element.classList.add('selected');
      selectedActivity = activity;
    }

    document.getElementById('submit-btn').addEventListener('click', () => {
      if (!selectedActivity) {
        alert("Please select an activity!");
        return;
      }

      document.getElementById('step-3').classList.add('hidden');
      document.getElementById('step-4').classList.remove('hidden');
      document.getElementById('summary-text').innerHTML = `DATE: ${selectedDate}<br><br>ACTIVITY: ${selectedActivity}`;

      sendNotification(selectedDate, selectedActivity);
    });

    function sendNotification(date, activity) {
      if (FORMSPREE_ENDPOINT === "YOUR_FORMSPREE_ENDPOINT_HERE") return;

      fetch(FORMSPREE_ENDPOINT, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          response: "YES!",
          chosen_date: date,
          chosen_activity: activity
        })
      });
    }
  </script>
